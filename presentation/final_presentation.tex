% === SLIDE 3: PROBLEM ===
\setsectionlabel{MOTIVATION}
\begin{frame}{The Communication Format Gap}
  \bodygraytext{Multi-agent LLM systems are increasingly built as pipelines: agents plan, execute, critique, retrieve, or integrate intermediate work.}

  \vspace{0.30cm}
  \bodygraytext{Most frameworks expose inter-agent communication as a flexible engineering choice, but do not answer a basic empirical question.}

  \vspace{0.42cm}
  \hspace{0.40cm}\pullquote{1.35}{Frameworks make the communication format flexible, but rarely tell us which format is efficient or reliable.}
\end{frame}
